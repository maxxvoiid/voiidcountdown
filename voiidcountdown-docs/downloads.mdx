const API = "https://api.github.com/repos/Voiid-Studios/voiidcountdown/releases?per_page=50";
const RELEASES_URL = "https://github.com/Voiid-Studios/voiidcountdown/releases";

function cmpSemver(a, b) {
  const pa = a.split(".").map(Number);
  const pb = b.split(".").map(Number);
  for (let i = 0; i < 3; i++) {
    if ((pa[i] || 0) > (pb[i] || 0)) return 1;
    if ((pa[i] || 0) < (pb[i] || 0)) return -1;
  }
  return 0;
}

function formatDate(iso) {
  try {
    return new Intl.DateTimeFormat("es-CL", {
      dateStyle: "medium",
      timeStyle: "short",
    }).format(new Date(iso));
  } catch {
    return iso;
  }
}

function pickAsset(assets) {
  if (!assets || !assets.length) return null;
  const preferred = assets.find((a) => /\.(jar|zip|mcaddon|mcpack)$/i.test(a.name));
  return preferred || assets[0];
}

function hydrateRelease(rel) {
  if (!rel) {
    return {
      version: "No release found",
      meta: "",
      downloadHref: RELEASES_URL,
      releaseHref: RELEASES_URL,
      buttonLabel: "View releases",
    };
  }

  const asset = pickAsset(rel.assets);
  return {
    version: `Version ${rel.tag_name}`,
    meta: `Published on ${formatDate(rel.published_at)} · ${rel.assets.length} asset(s)`,
    downloadHref: asset ? asset.browser_download_url : rel.html_url,
    releaseHref: rel.html_url,
    buttonLabel: asset ? "Download" : "View assets",
  };
}

function FeatureRow({ feature, lite, full }) {
