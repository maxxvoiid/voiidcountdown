function cmpSemver(a, b) {
  const pa = a.split(".").map(Number);
  const pb = b.split(".").map(Number);
  for (let i = 0; i < 3; i += 1) {
    if ((pa[i] || 0) > (pb[i] || 0)) return 1;
    if ((pa[i] || 0) < (pb[i] || 0)) return -1;
  }
  return 0;
}

function formatDate(iso) {
  try {
    return new Intl.DateTimeFormat("es-CL", { dateStyle: "medium", timeStyle: "short" }).format(new Date(iso));
  } catch (err) {
    return iso;
  }
}

function pickAsset(assets) {
  if (!assets || !assets.length) return null;
  const preferred = assets.find((asset) => /\.(jar|zip|mcaddon|mcpack)$/i.test(asset.name));
  return preferred || assets[0];
}

function hydratePlan(release) {
  if (!release) {
    return {
      versionText: "No se encontró una versión",
      metaText: "Revisa las publicaciones de GitHub para más detalles.",
      downloadHref: "https://github.com/Voiid-Studios/voiidcountdown/releases",
      releaseHref: "https://github.com/Voiid-Studios/voiidcountdown/releases",
      downloadLabel: "Ver lanzamientos",
    };
  }

  const asset = pickAsset(release.assets);
  return {
    versionText: `Versión ${release.tag_name}`,
    metaText: `Publicado el ${formatDate(release.published_at)} · ${release.assets.length} archivo(s)`,
    downloadHref: asset ? asset.browser_download_url : release.html_url,
    releaseHref: release.html_url,
    downloadLabel: asset ? "Download" : "Ver assets",
  };
}
