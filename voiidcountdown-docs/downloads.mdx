const API = "https://api.github.com/repos/Voiid-Studios/voiidcountdown/releases?per_page=50";
const DEFAULT_RELEASE_URL = "https://github.com/Voiid-Studios/voiidcountdown/releases";

const initialState = {
  versionText: "Cargando...",
  meta: "",
  downloadUrl: DEFAULT_RELEASE_URL,
  releaseUrl: DEFAULT_RELEASE_URL,
  downloadLabel: "Download",
};

const cmpSemver = (a, b) => {
  const pa = a.split(".").map(Number);
  const pb = b.split(".").map(Number);
  for (let i = 0; i < 3; i++) {
    if ((pa[i] || 0) > (pb[i] || 0)) return 1;
    if ((pa[i] || 0) < (pb[i] || 0)) return -1;
  }
  return 0;
};

const formatDate = (iso) => {
  try {
    return new Intl.DateTimeFormat("es-CL", {
      dateStyle: "medium",
      timeStyle: "short",
    }).format(new Date(iso));
  } catch (err) {
    return iso;
  }
};

const pickAsset = (assets) => {
  if (!assets || !assets.length) return null;
  const preferred = assets.find((asset) =>
    /\.(jar|zip|mcaddon|mcpack)$/i.test(asset.name)
  );
  return preferred || assets[0];
};

const hydrateRelease = (release) => {
  if (!release) {
    return {
      versionText: "No se encontró una versión",
      meta: "Revisa todas las publicaciones en GitHub.",
      downloadUrl: DEFAULT_RELEASE_URL,
      releaseUrl: DEFAULT_RELEASE_URL,
      downloadLabel: "Ver publicaciones",
    };
  }

  const asset = pickAsset(release.assets);

  return {
    versionText: `Versión ${release.tag_name}`,
    meta: `Publicado el ${formatDate(release.published_at)} · ${
      release.assets?.length ?? 0
    } archivo(s)`,
    downloadUrl: asset ? asset.browser_download_url : release.html_url,
    releaseUrl: release.html_url,
    downloadLabel: asset ? "Descargar" : "Ver assets",
  };
};
