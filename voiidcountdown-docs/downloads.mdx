function cmpSemver(a, b) {
  const pa = a.split(".").map(Number);
  const pb = b.split(".").map(Number);
  for (let i = 0; i < 3; i++) {
    if ((pa[i] || 0) > (pb[i] || 0)) return 1;
    if ((pa[i] || 0) < (pb[i] || 0)) return -1;
  }
  return 0;
}

function formatDate(iso) {
  try {
    return new Intl.DateTimeFormat("es-CL", {
      dateStyle: "medium",
      timeStyle: "short",
    }).format(new Date(iso));
  } catch {
    return iso;
  }
}

function pickAsset(assets) {
  if (!assets || !assets.length) return null;
  const preferred = assets.find((asset) => /\.(jar|zip|mcaddon|mcpack)$/i.test(asset.name));
  return preferred || assets[0];
}

function findLatest(releases, major) {
  const filtered = releases.filter((release) => release.tag_name.startsWith(`${major}.`));
  return filtered.sort((a, b) => cmpSemver(b.tag_name, a.tag_name))[0] || null;
}

function hydrateRelease(release) {
  if (!release) {
    return {
      versionText: "No release found",
      metaText: "",
      downloadUrl: FALLBACK_RELEASES_URL,
      releaseUrl: FALLBACK_RELEASES_URL,
      buttonLabel: "Download",
    };
  }

  const asset = pickAsset(release.assets);
  return {
    versionText: `Version ${release.tag_name}`,
    metaText: `Published on ${formatDate(release.published_at)} · ${release.assets.length} asset(s)`,
    downloadUrl: asset ? asset.browser_download_url : release.html_url,
    releaseUrl: release.html_url,
    buttonLabel: asset ? "Download" : "View assets",
  };
}
